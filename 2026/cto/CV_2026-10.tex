% A row is one unbreakable line: aside column, timeline rule (with an
% optional marker), body minipage. Rows are stacked without interline glue
% so the rule stays continuous; spacing lives in the bottom padding.

\newcommand{\cvrule}{\vrule width \rulewidth}
\newcommand{\cvrulestub}{\vrule width \rulewidth height 9pt depth 0pt}
\newcommand{\cvaside}[1]{%
  \parbox[t]{\dimexpr\asidewidth-\asidegap}{%
    \raggedleft\xsfont\leavevmode\color{muted}#1\par
    \vspace{\cvrowpad}}%
  \hspace{\asidegap}}

% \begin{cvrow}[bottom padding]{aside}{marker} body \end{cvrow}
\newenvironment{cvrow}[3][0pt]{%
  \par\nointerlineskip\noindent
  \def\cvrowpad{#1}%
  \cvaside{#2}%
  {\color{rule}\cvrule}#3\hspace{\bodygap}%
  \begin{minipage}[t]{\bodywidth}\raggedright}
 {\par\vspace{\cvrowpad}\hrule height 0pt\end{minipage}\par}

% \cvindent{body}: body column without the timeline (header, summary).
